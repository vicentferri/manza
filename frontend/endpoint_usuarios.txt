// ─── POST /sm/usuario ─────────────────────────────────────────────────────────
// Crea un nuevo usuario
router.post('/usuario', async (req, res) => {
  try {
    const { name, surname, email, password, empresa, subempresa, ambito } = req.body;

    // --- SQL Server ---
    await pool.request()
      .input('name',       sql.VarChar(100), name       || null)
      .input('surname',    sql.VarChar(100), surname    || null)
      .input('email',      sql.VarChar(255), email      || null)
      .input('password',   sql.VarChar(100), password   || null)
      .input('empresa',    sql.Int,          empresa    && empresa !== '-1' ? parseInt(empresa) : null)
      .input('subempresa', sql.Int,          subempresa && subempresa !== '-1' ? parseInt(subempresa) : null)
      .input('ambito',     sql.VarChar(10),  ambito     || 'E')
      .query(`
        INSERT INTO SOL_USUARIOS (name, surname, email, password, empresa, subempresa, ambito)
        VALUES (@name, @surname, @email, @password, @empresa, @subempresa, @ambito)
      `);

    // --- MySQL ---
    // await db.query(
    //   `INSERT INTO SOL_USUARIOS (name, surname, email, password, empresa, subempresa, ambito)
    //    VALUES (?, ?, ?, ?, ?, ?, ?)`,
    //   [name || null, surname || null, email || null, password || null,
    //    empresa && empresa !== '-1' ? empresa : null,
    //    subempresa && subempresa !== '-1' ? subempresa : null,
    //    ambito || 'E']
    // );

    return res.json({ message: 'OK' });

  } catch (err) {
    console.error('[POST /sm/usuario]', err);
    return res.status(500).json({ message: 'ERROR', detail: err.message });
  }
});


// ─── DELETE /sm/usuario/:id ───────────────────────────────────────────────────
// Elimina un usuario por su idrow
router.delete('/usuario/:id', async (req, res) => {
  try {
    const idrow = parseInt(req.params.id, 10);

    // --- SQL Server ---
    await pool.request()
      .input('idrow', sql.Int, idrow)
      .query('DELETE FROM SOL_USUARIOS WHERE idrow = @idrow');

    // --- MySQL ---
    // await db.query('DELETE FROM SOL_USUARIOS WHERE idrow = ?', [idrow]);

    return res.json({ message: 'OK' });

  } catch (err) {
    console.error('[DELETE /sm/usuario/:id]', err);
    return res.status(500).json({ message: 'ERROR', detail: err.message });
  }
});

-- NOTA: EDICIÓN DE CAMPOS Y PRODUCTOS --

  El frontend usa el endpoint genérico ya existente:

    POST /sm/update_table

  con body:
    {
      "table":  "SOL_USUARIOS",
      "column": "idrow",
      "id":     <idrow>,
      "field":  "<campo>",         (name|surname|email|empresa|subempresa|ambito|prod_1|...|prod_7)
      "value":  "<valor>"
    }

  El toggle inline de productos también usa /sm/update_table.
  La edición en bloque lanza múltiples llamadas a /sm/update_table
  en paralelo (una por usuario × campo con valor != '').

  IMPORTANTE: asegurarse de que la tabla SOL_USUARIOS está en
  la lista de tablas permitidas por el endpoint /sm/update_table.
